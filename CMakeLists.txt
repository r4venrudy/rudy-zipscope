cmake_minimum_required(VERSION 3.20)
project(rudy_zipscope VERSION 0.1.0 LANGUAGES C CXX)
option(ZIPSCOPE_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(ZIPSCOPE_REQUIRE_CLI_TESTS "Require independent Python command/oracle tests" OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(zipscope_inflate STATIC third_party/zlib/adler32.c third_party/zlib/crc32.c third_party/zlib/inflate.c third_party/zlib/inftrees.c third_party/zlib/inffast.c third_party/zlib/zutil.c)
target_include_directories(zipscope_inflate PUBLIC third_party/zlib)
add_library(zipscope_core src/model.cpp src/hash.cpp src/parser.cpp src/content.cpp src/names.cpp src/output.cpp src/compare.cpp src/io.cpp)
target_link_libraries(zipscope_core PRIVATE zipscope_inflate)
target_include_directories(zipscope_core PUBLIC include)
target_include_directories(zipscope_core SYSTEM PUBLIC third_party)
target_compile_features(zipscope_core PUBLIC cxx_std_20)
add_executable(zipscope_tests tests/tests.cpp)
target_link_libraries(zipscope_tests PRIVATE zipscope_core)
add_executable(zipscope src/main.cpp)
target_link_libraries(zipscope PRIVATE zipscope_core)
add_executable(zipscope_embed examples/embed.cpp)
target_link_libraries(zipscope_embed PRIVATE zipscope_core)
if(MINGW)
  target_link_options(zipscope PRIVATE -municode)
endif()
foreach(target zipscope_core zipscope_tests zipscope zipscope_embed)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    if(MINGW)
      target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    if(ZIPSCOPE_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
if(ZIPSCOPE_SANITIZE AND NOT MSVC)
  target_compile_options(zipscope_inflate PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
enable_testing()
add_test(NAME core-contracts COMMAND zipscope_tests)
add_test(NAME embedded-api COMMAND zipscope_embed)
if(ZIPSCOPE_REQUIRE_CLI_TESTS)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
else()
  find_package(Python3 3.9 COMPONENTS Interpreter)
endif()
if(Python3_FOUND)
  add_test(NAME independent-binary-oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/oracle.py $<TARGET_FILE:zipscope> ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME installed-command-contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:zipscope> ${CMAKE_CURRENT_SOURCE_DIR})
endif()
install(TARGETS zipscope RUNTIME DESTINATION bin)
install(FILES README.md LICENSE SECURITY.md CONTRIBUTING.md THIRD_PARTY.md DESTINATION .)
install(DIRECTORY docs assets examples benchmarks DESTINATION . PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES third_party/zlib/LICENSE DESTINATION licenses RENAME zlib-LICENSE)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-MIT.txt)
