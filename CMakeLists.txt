cmake_minimum_required(VERSION 3.20)
project(rudy_zipscope VERSION 0.1.0 LANGUAGES C CXX)
option(ZIPSCOPE_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(zipscope_core src/model.cpp)
target_include_directories(zipscope_core PUBLIC include)
target_compile_features(zipscope_core PUBLIC cxx_std_20)
add_executable(zipscope_tests tests/tests.cpp)
target_link_libraries(zipscope_tests PRIVATE zipscope_core)
foreach(target zipscope_core zipscope_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    if(MINGW)
      target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    if(ZIPSCOPE_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME core-contracts COMMAND zipscope_tests)
